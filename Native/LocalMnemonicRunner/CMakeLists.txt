cmake_minimum_required(VERSION 3.22)
project(LocalMnemonicRunner LANGUAGES C CXX)
if(NOT APPLE)
  message(FATAL_ERROR "LocalMnemonicRunner requires macOS Seatbelt")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT LLAMA_SOURCE_DIR)
  message(FATAL_ERROR "Pass the pinned LLAMA_SOURCE_DIR")
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
include("${CMAKE_CURRENT_SOURCE_DIR}/EmbedMetal.cmake")
add_executable(LocalMnemonicRunner main.cpp)
target_link_libraries(LocalMnemonicRunner PRIVATE llama)
target_compile_options(LocalMnemonicRunner PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
add_executable(LocalMnemonicBenchmark main.cpp)
target_compile_definitions(LocalMnemonicBenchmark PRIVATE LOCAL_AI_BENCHMARK=1)
target_link_libraries(LocalMnemonicBenchmark PRIVATE llama)
target_compile_options(LocalMnemonicBenchmark PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
add_executable(LocalMnemonicSandboxProbe SandboxProbe.cpp)
target_compile_options(LocalMnemonicSandboxProbe PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
