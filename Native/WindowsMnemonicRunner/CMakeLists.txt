cmake_minimum_required(VERSION 3.22)
project(WindowsMnemonicRunner LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "WindowsMnemonicRunner requires Windows AppContainer")
endif()
if(NOT LLAMA_SOURCE_DIR)
  message(FATAL_ERROR "Pass the pinned LLAMA_SOURCE_DIR")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static audited runtime" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "No dynamic backend lookup" FORCE)
set(GGML_CUDA ON CACHE BOOL "GPU-only dedicated-memory inference" FORCE)
set(GGML_STATIC ON CACHE BOOL "Static CUDA runtime; audited cuBLAS DLLs" FORCE)
set(GGML_CUDA_NO_VMM ON CACHE BOOL "Use dedicated cudaMalloc buffers" FORCE)
set(GGML_CUDA_NO_PEER_COPY ON CACHE BOOL "Single dedicated GPU" FORCE)
set(GGML_CUDA_GRAPHS OFF CACHE BOOL "Placement checks before graph operations" FORCE)
set(GGML_CUDA_FA ON CACHE BOOL "GPU Flash Attention" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "No graphics driver permissions" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "No dependency DLLs" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "No HTTP" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_executable(WindowsMnemonicRunner main.cpp)
target_link_libraries(WindowsMnemonicRunner PRIVATE llama ws2_32 advapi32 bcrypt)
target_include_directories(WindowsMnemonicRunner PRIVATE "${LLAMA_SOURCE_DIR}/src")
target_compile_definitions(WindowsMnemonicRunner PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(WindowsMnemonicRunner PRIVATE /W4 /GS /guard:cf /sdl)
target_link_options(WindowsMnemonicRunner PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
